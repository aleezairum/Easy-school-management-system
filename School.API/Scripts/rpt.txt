
CREATE PROCEDURE [dbo].[SpRPT_FeeVoucher]
(
    @VID INT 
)
AS
BEGIN

SET NOCOUNT ON;

SELECT
    H.VID, H.VNo,  H.VDate, H.PaymentMethodID, PM.VName AS PaymentMethod, H.ParentID, P.FatherName AS ParentName, H.BalanceAmount, H.PaidAmount, H.Remarks
	,H.InsertedBy, H.InsertedDate, H.InsertedIp, H.UpdatedBy, H.UpdatedDate, H.UpdatedIp,D.StudentFeeID, D.PaidAmount DetailPaidAmount, D.PendingBalance
	, SF.VDate FeeMonth, SF.StudentID, SF.SessionID, SF.SectionID, SF.GradeID, SF.FeeTypeID, SF.FeeAmount, SF.ClearedAmount, SF.Remarks ApplyFeeRemarks
	, S.Name AS StudentName, C.VName + ' (' + Sec.VName + ')' AS SectionName, FT.VName FeeTypeName, SU.UserFullName

FROM FeeVoucherMain H
INNER JOIN FeeVoucherDetail D ON H.VID = D.MID 
INNER JOIN SMSStudentFee SF ON D.StudentFeeID = SF.VID
INNER JOIN SMSStudent S ON SF.StudentID = S.VID 
INNER JOIN SMSSection Sec ON SF.SectionID = Sec.VID
INNER JOIN SMSClass C ON Sec.ClassID = C.VID
INNER JOIN AcademicGrades G ON SF.GradeID = G.VID
INNER JOIN SMSFeeType FT ON SF.FeeTypeID = FT.VID
LEFT JOIN PaymentMethod PM ON H.PaymentMethodID = PM.VID
LEFT JOIN SMSParent P ON H.ParentID = P.VID
INNER JOIN SecUser SU ON SU.VID = H.InsertedBy
WHERE H.VID = @VID AND H.IsActive = 1

ORDER BY H.VDate DESC

END
